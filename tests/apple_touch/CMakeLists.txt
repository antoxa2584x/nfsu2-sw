cmake_minimum_required(VERSION 3.20)
project(apple_touch_tests C)
set(CMAKE_C_STANDARD 11)
find_package(SDL2 REQUIRED)
add_executable(touch_input test_input.c)
target_link_libraries(touch_input PRIVATE SDL2::SDL2)
add_executable(frame_pacing test_pacing.c)
add_executable(game_state test_game_state.c ../../src/apple/apple_game_state.c)
foreach(target touch_input frame_pacing game_state)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endforeach()
enable_testing()
add_test(NAME touch_input COMMAND touch_input)
add_test(NAME game_state COMMAND game_state)
add_test(NAME frame_pacing_unlimited COMMAND frame_pacing)
add_test(NAME frame_pacing_60 COMMAND frame_pacing)
set_tests_properties(frame_pacing_60 PROPERTIES ENVIRONMENT "NFSU2_FPS_LIMIT=60")
add_test(NAME frame_pacing_30 COMMAND frame_pacing)
set_tests_properties(frame_pacing_30 PROPERTIES ENVIRONMENT "NFSU2_FPS_LIMIT=30")
add_test(NAME frame_pacing_invalid COMMAND frame_pacing)
set_tests_properties(frame_pacing_invalid PROPERTIES ENVIRONMENT "NFSU2_FPS_LIMIT=invalid")
add_test(NAME frame_pacing_120 COMMAND frame_pacing)
set_tests_properties(frame_pacing_120 PROPERTIES ENVIRONMENT "NFSU2_FPS_LIMIT=120")
